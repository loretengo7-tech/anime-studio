name: Deploy to GitHub Pages

# ANIME STUDIO — build CI desde el código fuente y deploy con UN solo
# artifact github-pages por run (regla de la especificación).
on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build-deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Bun
        uses: oven-sh/setup-bun@v2
        with:
          bun-version: latest

      - name: Install dependencies
        run: bun install --frozen-lockfile || bun install

      - name: Build static export (base path /anime-studio)
        run: NEXT_PUBLIC_BASE_PATH=/anime-studio bash scripts/build-static.sh

      - name: Verify build
        run: |
          test -f dist-static/index.html || { echo "::error::dist-static sin index.html"; exit 1; }
          test -f dist-static/sw.js || { echo "::error::dist-static sin sw.js"; exit 1; }
          test -f dist-static/.nojekyll || { echo "::error::falta .nojekyll"; exit 1; }
          echo "CACHE_VERSION desplegada: $(grep -o "as-v[0-9]*" dist-static/sw.js | head -1)"

      - name: Configure Pages (auto-enable)
        uses: actions/configure-pages@v5
        with:
          enablement: true

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: dist-static

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
